cmake_minimum_required(VERSION 3.20)
project(HerdMeCore VERSION 0.1.0 LANGUAGES CXX)

add_library(herdme-core-library src/core.cpp)
target_include_directories(herdme-core-library PUBLIC include)
target_compile_features(herdme-core-library PUBLIC cxx_std_20)

add_executable(herdme-core src/main.cpp)
target_link_libraries(herdme-core PRIVATE herdme-core-library)

include(CTest)
if(BUILD_TESTING)
    add_executable(herdme-core-tests tests/core_tests.cpp)
    target_link_libraries(herdme-core-tests PRIVATE herdme-core-library)
    add_test(NAME herdme-core-tests COMMAND herdme-core-tests)
endif()

if(MSVC)
    target_compile_options(herdme-core-library PRIVATE /W4 /permissive-)
    target_compile_options(herdme-core PRIVATE /W4 /permissive-)
    if(BUILD_TESTING)
        target_compile_options(herdme-core-tests PRIVATE /W4 /permissive-)
    endif()
else()
    target_compile_options(herdme-core-library PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(herdme-core PRIVATE -Wall -Wextra -Wpedantic)
    if(BUILD_TESTING)
        target_compile_options(herdme-core-tests PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endif()
